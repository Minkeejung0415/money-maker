def _platt(raw_probs, y):
    import numpy as np
    from sklearn.linear_model import LogisticRegression
    p = np.clip(raw_probs, 1e-6, 1-1e-6)
    logits = np.log(p/(1-p)).reshape(-1, 1)
    return LogisticRegression().fit(logits, y)

def calibrated(calibrator, raw_probs):

    Module docstring: "Leakage-safe EPL pregame features shared by training and live inference."

    EPL_FEATURE_NAMES tuple (14 features, fixed order):
      ("home_xg_for", "home_xg_against", "away_xg_for", "away_xg_against",
       "home_form_points", "home_form_goal_diff", "away_form_points", "away_form_goal_diff",
       "h2h_home_win_rate", "home_rest_days", "away_rest_days",
       "home_corners_pg", "away_corners_pg", "home_aerial_pct")

    EPLTeamState dataclass (mirrors mlb_training.TeamState but soccer-specific):
      games: int = 0
      wins: int = 0
      xg_for_total: float = 0.0
      xg_against_total: float = 0.0
      elo: float = 1500.0
      last_date: str = ""
      form_buffer: list = field(default_factory=list)
      # Each entry: {"points": int (W=3,D=1,L=0), "goal_diff": float}
      # Kept to last FORM_WINDOW*2 entries to bound memory; sliced to last 5 when computing features.

      Properties:
        xg_for_pg: float — xg_for_total / games if games else 1.5
        xg_against_pg: float — xg_against_total / games if games else 1.5
        form_points_n(n=5): float — sum of points across last n entries in form_buffer (0.0 if empty)
        form_goal_diff_n(n=5): float — sum of goal_diffs across last n entries in form_buffer (0.0 if empty)

    FORM_WINDOW: int = 5  (module-level constant)

    Internal helper _days_rest(last_date: str, game_date: str) -> float:
      - Mirror mlb_training._days_rest exactly: max(0, min(7, delta.days - 1)), default 3.0 when last_date is empty.

    Internal helper _feature_vector(home: EPLTeamState, away: EPLTeamState, game_date: str,
                                     h2h_home_win_rate: float,
                                     fbref_home: dict, fbref_away: dict) -> dict[str, float]:
      Returns dict with all 14 EPL_FEATURE_NAMES keys.
      Feature values:
        home_xg_for: home.xg_for_pg
        home_xg_against: home.xg_against_pg
        away_xg_for: away.xg_for_pg
        away_xg_against: away.xg_against_pg
        home_form_points: home.form_points_n(FORM_WINDOW)
        home_form_goal_diff: home.form_goal_diff_n(FORM_WINDOW)
        away_form_points: away.form_points_n(FORM_WINDOW)
        away_form_goal_diff: away.form_goal_diff_n(FORM_WINDOW)
        h2h_home_win_rate: h2h_home_win_rate
        home_rest_days: _days_rest(home.last_date, game_date)
        away_rest_days: _days_rest(away.last_date, game_date)

        away_corners_pg: float(fbref_away.get("corners_per_game", 0.0))
        home_aerial_pct: float(fbref_home.get("aerial_won_pct", 0.0))

    Public function build_epl_pregame_rows(games: list[dict],
                                            xg_map: dict[str, dict] | None = None,
                                            fbref_map: dict[str, dict] | None = None
                                           ) -> tuple[list[dict], dict[str, dict]]:
      - xg_map: optional {team_name: {"xG_for": float, "xG_against": float}} from soccer_stats.py
      - fbref_map: optional {team_name: {"corners_per_game": float, "aerial_won_pct": float}} from get_fbref_set_pieces()
      - Sort games chronologically by (date, game_id).
      - Maintain TWO separate state structures:
        (a) states: dict[str, EPLTeamState] — per-team xG/Elo/form rolling state
        (b) h2h_log: dict[str, list] — per-matchup H2H history, keyed by f"{team_a}|{team_b}"
      - For each game (ALL finished games, including draws):
          * Extract PREGAME features from current state (leakage-safe: features before this game's result).
          * Compute h2h_home_win_rate from h2h_log BEFORE updating it:
              h2h_key = f"{home}|{away}"
              both_directions = h2h_log.get(h2h_key, []) + h2h_log.get(f"{away}|{home}", [])
              recent_h2h = sorted(both_directions, key=lambda r: r["date"])[-FORM_WINDOW:]
              if recent_h2h:
                  home_side_wins = sum(1 for r in recent_h2h if r["home"] == home and r["winner"] == "home")
                  h2h_home_win_rate = home_side_wins / len(recent_h2h)
              else:
                  h2h_home_win_rate = 0.5
          * fbref_home = (fbref_map or {}).get(home, {})
          * fbref_away = (fbref_map or {}).get(away, {})
          * features = _feature_vector(h, a, game_date, h2h_home_win_rate, fbref_home, fbref_away)
          * If home_score == away_score (draw): still update form/H2H state below, but DO NOT append to rows
            (binary classifier skips draws — same as mlb_training).
          * If NOT a draw: append {"date": game_date, "home_team": home, "away_team": away,
                                   **features, "home_win": int(float(hs) > float(aws))} to rows.
          * Update state AFTER extracting features:
              home_win = float(hs) > float(aws)
              draw = float(hs) == float(aws)
              gd = float(hs) - float(aws)
              points_h = 3 if home_win else (1 if draw else 0)
              points_a = 3 if not home_win and not draw else (1 if draw else 0)
              h.form_buffer.append({"points": points_h, "goal_diff": gd})
              a.form_buffer.append({"points": points_a, "goal_diff": -gd})
              h.form_buffer = h.form_buffer[-(FORM_WINDOW * 2):]  # bound memory
              a.form_buffer = a.form_buffer[-(FORM_WINDOW * 2):]
              Elo update (K=20, same formula as mlb_training): expected, change, h.elo += change, a.elo -= change
              Accumulate xg_for/against from game dict "home_xg"/"away_xg" if present (else skip).
              h.games += 1; a.games += 1
              if home_win: h.wins += 1
              if not home_win and not draw: a.wins += 1
              h.last_date = game_date; a.last_date = game_date
              h2h_log.setdefault(h2h_key, []).append({
                  "date": game_date, "home": home, "away": away,
                  "winner": "home" if home_win else ("draw" if draw else "away")
              })
      - Return (rows, {team: asdict(state) for team, state in states.items()})
